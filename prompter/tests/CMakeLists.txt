# Serbian catalog RESOLUTION: proves the sr Cyrillic-primary
# catalog actually resolves at runtime through ki18n — not merely that the .po
# exists. Compile the sr catalog into a PRIVATE XDG data tree laid out exactly
# like a system install (<root>/locale/sr/LC_MESSAGES/<domain>.mo); the test
# prepends <root> to XDG_DATA_DIRS so KLocalizedString finds it through the
# standard QStandardPaths::GenericDataLocation lookup (stable across ki18n
# versions, no system-wide install, no version-fragile addDomainLocaleDir).
find_program(MSGFMT_EXECUTABLE NAMES msgfmt)
if(MSGFMT_EXECUTABLE)
    set(_cat_locale_root "${CMAKE_CURRENT_BINARY_DIR}/test-locale")
    set(_cat_sr_mo "${_cat_locale_root}/locale/sr/LC_MESSAGES/librescrs-pinentry-kde.mo")
    add_custom_command(
        OUTPUT "${_cat_sr_mo}"
        COMMAND ${CMAKE_COMMAND} -E make_directory "${_cat_locale_root}/locale/sr/LC_MESSAGES"
        COMMAND ${MSGFMT_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/../po/sr/librescrs-pinentry-kde.po" -o "${_cat_sr_mo}"
        DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/../po/sr/librescrs-pinentry-kde.po"
        COMMENT "Compiling sr prompter catalog for PromptCatalogResolutionTest"
        VERBATIM)
    add_custom_target(PromptCatalogResolutionTest_catalog DEPENDS "${_cat_sr_mo}")

    add_executable(PromptCatalogResolutionTest PromptCatalogResolutionTest.cpp)
    add_dependencies(PromptCatalogResolutionTest PromptCatalogResolutionTest_catalog)
    target_link_libraries(PromptCatalogResolutionTest PRIVATE
        Qt6::Core
        KF6::I18n
        GTest::gtest)
    add_test(NAME PromptCatalogResolutionTest COMMAND PromptCatalogResolutionTest)
    # LANG MUST name a real UTF-8 locale: ki18n resolves catalogs through the C
    # library's gettext, which suppresses translation entirely under the
    # C / POSIX / C.UTF-8 locale (so the test would silently pass as English and
    # hide a broken sr catalog). LANGUAGE=sr then selects Serbian. CI must have
    # en_US.UTF-8 generated (see the workflow's locale step); a developer desktop
    # already carries a UTF-8 locale.
    # XDG_DATA_HOME is PINNED at the test tree, not merely prepended to
    # XDG_DATA_DIRS. QStandardPaths::GenericDataLocation searches XDG_DATA_HOME
    # FIRST, so on a machine that has dogfood-installed the prompter into
    # ~/.local/share the test reads that catalog instead of the one just built —
    # and passes or fails on whatever was installed last. Found by adding a
    # string and watching the test resolve the old catalog without it.
    set_tests_properties(PromptCatalogResolutionTest PROPERTIES
        ENVIRONMENT "QT_QPA_PLATFORM=offscreen;LANGUAGE=sr;LANG=en_US.UTF-8;XDG_DATA_HOME=${_cat_locale_root}"
        ENVIRONMENT_MODIFICATION "XDG_DATA_DIRS=path_list_prepend:${_cat_locale_root}")
else()
    message(WARNING "msgfmt not found — PromptCatalogResolutionTest (Serbian catalog resolution) is skipped")
endif()

# own pathological worst case is 25 consecutive 5 s pump budgets, about 125 s.
